		}

				}
			}
		}
	}

			factor = 0.5
			NOT = {
				estate_loyalty = {
					estate = estate_workers
					loyalty = 60
				}
			}
			is_at_war = no
		}

			factor = 0.5
			NOT = {
				estate_loyalty = {
					estate = estate_workers
					loyalty = 50
				}
			}

			factor = 1

				estate_loyalty = {

					loyalty = 30

				NOT = { has_country_flag = estate_workers_in_power }
