# Government - Dwarven Outposts

reached_dwarven_outpost_slot_limit = {
	variable_arithmetic_trigger = {
		export_to_variable = {
			variable_name = current_amount_of_dwarven_outposts
			value = modifier:dwarven_outpost_amount_check_influence
		}
		export_to_variable = {
			variable_name = max_amount_of_dwarven_outposts
			value = modifier:num_of_dwarven_outposts
		}
		check_variable = {
			which = current_amount_of_dwarven_outposts
			which = max_amount_of_dwarven_outposts
		}
	}
}

enough_dwarven_outpost_slots = {
	variable_arithmetic_trigger = {
		export_to_variable = {
			variable_name = current_amount_of_dwarven_outposts
			value = modifier:dwarven_outposts_amount_check_influence
		}
		export_to_variable = {
			variable_name = max_amount_of_dwarven_outposts
			value = modifier:num_of_dwarven_outposts
		}
		NOT = {
			check_variable = {
				which = current_amount_of_dwarven_outposts
				which = max_amount_of_dwarven_outposts
			}
		}
	}
}
