# Describes how religion is spread to new provinces from
# Centers of Reformation and the "Propagate Religion" Trading Policy

propagate_religion_policy = {
	
	trading_policy = yes # Religion is taken from country using the trading policy.	max_distance is not used.

	map_icon = GFX_religious_conversion_center_trading_policy

	base_conversion_speed = 2.0
	
	target_province_weights = {
	
		#Will only target provinces within same node
		#From Scope is country using the trading policy
		
		factor = 5
				
		modifier = {
			factor = 0
			has_owner_religion = yes
			owner = {
				has_country_flag = immunity_against_propogate_religion
			}
		}
		modifier = {
			factor = 0
			has_missionary = yes
		}
		modifier = {
			factor = 0
			is_empty = yes
		}
		modifier = {
			factor = 0
			has_province_modifier = religious_center
		}
		modifier = {
			factor = 0
			has_province_modifier = religious_zeal_at_conv
		}
		modifier = {
			factor = 0
			OR = {
				religion_group = chaos_group
				religion_group = order_group
				religion_group = autocratic_group
				religion_group = hedonistic_group
				religion_group = transcendent_group
				religion_group = elder_god_group
				religion_group = horror_group
				religion_group = karamjan_group
				religion_group = menaphite_group
				religion_group = celestial_group
				religion_group = nonintervention_group
			}
			FROM = {
				NOT = { has_country_flag = can_propagate_religion_in_misc_religions #break this up into multiple if you want to use it }
			}
		}
		modifier = {
			factor = 0.25
			has_port = no
		}
		modifier = {
			factor = 0.5
			is_part_of_hre = yes
		}
		modifier = {
			factor = 1.15
			has_province_modifier = trading_settlement
		}
		modifier = {
			factor = 2
			OR = {
				any_neighbor_province = {
					religion = FROM
				}
				has_port = yes
			}
		}
		modifier = {
			factor = 1.5
			owner = {
				religion = FROM
			}
		}
		modifier = {
			factor = 1.25
			owner = {
				religion_group = FROM
			}
		}
	}
}
