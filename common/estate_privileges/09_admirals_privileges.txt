		}
	}

					value = 1
				}
			}
		}
	}
	on_granted = {
		if = {
			limit = {
				fishing_spot = 1
			}

		}

			years = 8
			trigger = {

			}

		}
		custom_tooltip = estate_admirals_monopoly_of_fish_effect_tooltip
	}
	on_granted_province = {
		if = {
			limit = {
				trade_goods = fishing_spot
			}
			add_province_modifier = {
				name = estate_admirals_monopoly_of_fish_mod
				duration = -1
			}
		}
		else_if = {
			limit = {
				has_province_modifier = estate_admirals_monopoly_of_fish_mod
			}
			remove_province_modifier = estate_admirals_monopoly_of_fish_mod
		}
	}
	on_revoked = {
		custom_tooltip = estate_admirals_monopoly_of_fish_revoke_tooltip
	}
	on_revoked_province = {
		if = {
			limit = {
				has_province_modifier = estate_admirals_monopoly_of_fish_mod
			}
			remove_province_modifier = estate_admirals_monopoly_of_fish_mod
		}
	}
	on_invalid = {
		custom_tooltip = estate_admirals_monopoly_of_fish_revoke_tooltip

	on_invalid_province = {

			limit = {
				has_province_modifier = estate_admirals_monopoly_of_fish_mod

			remove_province_modifier = estate_admirals_monopoly_of_fish_mod
