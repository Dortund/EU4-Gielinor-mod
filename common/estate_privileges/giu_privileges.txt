estate_nobles_gunpowder_monopoly = {
	icon = privilege_nobles_gunpowder_monopoly
	max_absolutism = -10
	conditional_modifier = { trigger = { has_absolutism_reduction_for_estate_privileges = yes } modifier = { max_absolutism = 2 } }
	loyalty = 0.05
	is_valid = {
		any_owned_province = {
			trade_goods = gunpowder
		}
	}
	can_select = {
		custom_trigger_tooltip = {
			tooltip = unavailable_during_revolution_disaster
			NOT = { has_disaster = revolution }
			NOT = {
				has_disaster_progress = {
					disaster = revolution
					value = 1
				}
			}
		}
	}
	on_granted = {
		if = {
			limit = {
				gunpowder = 1
			}
			add_mercantilism = 1
		}
		add_years_of_owned_provinces_production_income = {
			years = 8
			trigger = {
				trade_goods = gunpowder
			}
			custom_tooltip = MONOPOLY_INCOME_GUNPOWDER_DESC
		}
		custom_tooltip = estate_nobles_monopoly_of_gunpowder_effect_tooltip
	}
	on_granted_province = {
		if = {
			limit = {
				trade_goods = gunpowder
			}
			add_province_modifier = {
				name = estate_nobles_monopoly_of_gunpowder_mod
				duration = -1
			}
		}
		else_if = {
			limit = {
				has_province_modifier = estate_nobles_monopoly_of_gunpowder_mod
			}
			remove_province_modifier = estate_nobles_monopoly_of_gunpowder_mod
		}
	}
	on_revoked = {
		custom_tooltip = estate_nobles_monopoly_of_gunpowder_revoke_tooltip
	}
	on_revoked_province = {
		if = {
			limit = {
				has_province_modifier = estate_nobles_monopoly_of_gunpowder_mod
			}
			remove_province_modifier = estate_nobles_monopoly_of_gunpowder_mod
		}
	}
	on_invalid = {
		custom_tooltip = estate_nobles_monopoly_of_gunpowder_revoke_tooltip
	}
	on_invalid_province = {
		if = {
			limit = {
				has_province_modifier = estate_nobles_monopoly_of_gunpowder_mod
			}
			remove_province_modifier = estate_nobles_monopoly_of_gunpowder_mod
		}
	}
	penalties = {
	}
	benefits = {
	}

	cooldown_years = 10
	conditional_modifier = {
		trigger = {
			has_country_flag = buffed_monopolies
		}
		modifier = {
			max_absolutism = 10
			nobles_privilege_slots = 1
		}
	}
	on_cooldown_expires = {
		country_event = { id = giu_monopoly_privileges.6 }
	}
	ai_will_do = {
		factor = 0
	}
}
