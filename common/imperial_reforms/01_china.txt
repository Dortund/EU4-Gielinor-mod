kem_desert_recovery_reform = {
	empire = celestial_empire
	trigger = {
		empire_of_china_reform_passed = kem_expand_settlements_reform
		NOT = {
			any_province = {
				religion = amascut_religion
			}
		}
		custom_trigger_tooltip = {
			tooltip = kem_desert_recovery_capital_is_prosperous
			capital_scope = {
				is_prosperous = yes
			}
		}
	}
	member = {
		colony_cost_modifier = -0.1
	}
	emperor = {
		global_monthly_devastation = -0.1
	}
	on_effect = {
		if = {
			limit = {
				NOT = { has_global_flag = kem_expand_settlements_flag }
			}
			custom_tooltip = kem_desert_recovery_colonising_spree
			hidden_effect = {
				set_global_flag = kem_expand_settlements_flag
				every_subject_country = {
					limit = {
						capital_scope = {
							region = kharidian_desert_region
						}
						any_owned_province = {
							region = kharidian_desert_region
							has_empty_adjacent_province = yes
						}
					}
					country_event = { 
						id = empire_of_kharid_events.1
					}
				}
			}
		}
	}
	off_effect = {
	}
}
