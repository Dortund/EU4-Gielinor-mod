######################################################################
# Varlamore Loyalist rebels
######################################################################
varlamore_loyalist_rebels = {
	
	color = { 40 140 210 }
	
	area = nation 					# Stays within the nation
	government = any
	defection = none				# Will defect to the "best" alternative.
	independence = none				# Does not strive for independence
	unit_transfer = yes
	gfx_type = culture_province
	
	resilient = no					# Disbands on retreat
	reinforcing = no				# Does not receive reinforcements
	general = yes					# Does not use generals
	smart = yes						# Does not assault forts, tactically withdraw from battle, etc.
	disband_on_leader_death = no	# Pretender rebels live forever!
	will_relocate = yes				# Units will relocate if they have sieged all provinces nearby and has to move over sea.
	dynasty = yes					#same dynasty for rebel uprising
	
	artillery = 0.6
	infantry = 1.0
	cavalry = 0.2
	
	
	# Possible handle actions
	handle_action_negotiate = no
	handle_action_stability = yes
	handle_action_build_core = yes
	handle_action_send_missionary = yes		
	
	# The rebel type with the highest modifier for this province gets picked
	spawn_chance = {
		factor = 25
		modifier = {
			factor = 0
			owner = { NOT = { has_disaster = war_of_betrayal } }
		}
		modifier = {
			factor = 0
			owner = { has_country_flag = war_of_betrayal_loyalist_flag }
		}
	}
 	# This is checked for EACH province in the Area of Operations
	movement_evaluation = {
		factor = 1
		modifier = {
			factor = 0.1
			units_in_province = 1
		}
		modifier = {
			factor = 0.001
			controlled_by = REB
		}
		modifier = {
			factor = 4.0
			is_capital = yes
		}
		modifier = {
			factor = 1.5
			unrest = 2
		}
		modifier = {
			factor = 1.5
			unrest = 4
		}
		modifier = {
			factor = 1.5
			unrest = 6
		}
		modifier = {
			factor = 10
			controlled_by = KOU
		}
	}
	
 	# Province Scope
 	siege_won_trigger = {
 		NOT = { local_autonomy = 50 }
 	}
	siege_won_effect = {
		add_local_autonomy = 10	
	}
	
	# Country scope
	can_negotiate_trigger = {
		always = no
	}
	
	# Country scope
	can_enforce_trigger = {
		always = yes
	}
	
	# Localisation for their demands
	demands_description = "varlamore_loyalist_rebels_demand"
	
	# Country scope
	demands_enforced_effect = {
		hidden_effect = {
			add_country_modifier = {
				name = just_lost_to_rebels
				duration = 10
				hidden = yes
			}
			set_country_flag = var_loyalist_rebel_victory
		}
		custom_tooltip = var_loyalists_win_war_tooltip
		add_stability = -2
	}
}
