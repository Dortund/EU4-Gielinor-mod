country_event = {
	id = corruption_of_amascut.1
	title = corruption_of_amascut.1.t #Corruption of Amascut
	desc = corruption_of_amascut.1.d #As corruption has grown in [Root.GetName], so too has Amascut's power. 	picture = CORRUPTION_OF_AMASCUT_eventPicture

	goto = amascut_invasion_city

		}

		}

			factor = 0
		}
	}

		name = corruption_of_amascut.1.b #We must fight for our souls!
		add_stability = -2
	 	add_corruption = -1
		event_target:amascut_invasion_city = {
			amascut_religion_rebels = 3
		}
		ai_chance = { factor = 1 }
	}
}
