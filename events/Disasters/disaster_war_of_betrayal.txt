					type = great_project_temple_of_twilight
					tier = -2

			limit = {
				503 = {
					has_great_project = {
						type = great_project_temple_of_twilight
						tier = 1
					}
				}
			}

			503 = {
				add_great_project_tier = {
					type = great_project_temple_of_twilight

		NOT = {
			had_country_flag = { flag = VAR_war_of_betrayal_start days = 3650 }
		}
		any_country = {
			AND = {
				OR = {
					primary_culture = lovakengj
					culture_group = kourend_group
				}
				OR = {
					is_in_war = {
						attacker_leader = ROOT
						defender_leader = PREV
					}
					is_in_war = {
						attacker_leader = PREV
						defender_leader = ROOT
					}
				}
				NOT = {
					war_score_against = {
						who = ROOT
						value = 25
					}
				}
				NOT = {
					has_country_flag = VAR_declined_to_end_war_flag
				}
			}
		}

	}

	option = {

				id = war_of_betrayal.5
				days = 1
			}
		}
	}
	option = {
		name = war_of_betrayal.4.b #We cannot afford to make such an offer.
