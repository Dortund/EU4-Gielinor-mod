# Embers of [embers_province.GetName]
country_event = {
	id = queen_of_ashes_events.1
	title = queen_of_ashes_events.1.t
	desc = queen_of_ashes_events.1.d #In the devastated landscape of [embers_province.GetName], [Root.Monarch.GetTitle] [Root.Monarch.GetName] stares into the smouldering embers. The fading crackles almost seem to be saying something...
	picture = QUEEN_OF_ASHES_FALLEN_NIHIL_eventPicture
	
	trigger = {
		NOT = {
			religion = queen_of_ashes
		}
		NOT = {
			has_ruler_modifier = queen_of_ashes_intro_contemplating_flame
		}
		NOT = {
			has_country_flag = part_way_to_queen_of_ashes_flag
		}
		calc_true_if = {
			all_owned_province = {
				has_province_modifier = province_razed
			}
			amount = 12
		}
	}
	
	mean_time_to_happen = {
		months = 360
	}
	
	immediate = {
		random_owned_province = {
			limit = {
				has_province_modifier = province_razed
			}
			save_event_target_as = embers_province
		}
	}
	
	option = {
		name = queen_of_ashes_events.1.a #The fire speaks!
		custom_tooltip = queen_of_ashes_introduction_tooltip
		add_ruler_modifier = { 
			name = queen_of_ashes_intro_contemplating_flame
			duration = -1
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 10
				current_age = age_of_divination
			}
		}
		hidden_effect = {
			country_event = { 
				id = queen_of_ashes_events.2
				days = 1825
			}
		}
	}
	option = {
		name = queen_of_ashes_events.1.b #We should put those embers out.
		event_target:embers_province = {
			add_devastation = -5
		}		
		ai_chance = {
			factor = 10
			modifier = {
				factor = 0.1
				current_age = age_of_divination
			}
		}
	}
}

	desc = {
	   trigger = { NOT = { religion = amascut_religion } }
	   desc = queen_of_ashes_events.2.da #In [Root.Monarch.GetHerHis] chambers, a voice speaks to [Root.Monarch.GetTitle] [Root.Monarch.GetName] from a fireplace.\n\n§OYour devotion to the flame has not gone unnoticed.§!\n§OI am an emissary of the Queen of Ashes - a redeemer and life-giver.§!\n§OIf you are willing, I can enlighten you on her teachings.§!
	}
	desc = {
	   trigger = { religion = amascut_religion }
	   desc = queen_of_ashes_events.2.db #In [Root.Monarch.GetHerHis] chambers, a voice speaks to [Root.Monarch.GetTitle] [Root.Monarch.GetName] from a fireplace.\n\n§OYour devotion to the flame has not gone unnoticed.§!\n§OI am an emissary of the Queen of Ashes--\n\nSuddenly, the voice cuts out as the sound of Amascut screams into [Root.Monarch.GetTitle] [Root.Monarch.GetName]'s head.\n\n§RHow dare you defy my will!§!
	}

	picture = QUEEN_OF_ASHES_FALLEN_NIHIL_eventPicture
	
	trigger = {
		NOT = {
			religion = queen_of_ashes
		}
		has_ruler_modifier = queen_of_ashes_intro_contemplating_flame
	}
	
	is_triggered_only = yes
	
	immediate = {
	}
	
	option = {
		name = queen_of_ashes_events.2.a #A voice from a fire told me to join their religion? Sign me up!
		trigger = {
			NOT = { religion = amascut_religion }
		}
		remove_country_modifier = queen_of_ashes_intro_contemplating_flame
		set_country_flag = part_way_to_queen_of_ashes_flag
		set_ruler_religion = queen_of_ashes
		if = {
			limit = {
				has_heir = yes
			}
			set_heir_religion = queen_of_ashes
		}
		custom_tooltip = queen_of_ashes_progression_tooltip
		add_stability = -1
		hidden_effect = {
			country_event = { 
				id = queen_of_ashes_events.3
				days = 1825
			}
		}
		ai_chance = {
			factor = 1
		}
	}
	option = {
		name = queen_of_ashes_events.2.b #Ashes are dropped on the floor all the time! They hardly deserve worship.
		trigger = {
			NOT = { religion = amascut_religion }
		}
		add_prestige = 10
		ai_chance = {
			factor = 0
		}
	}
	option = {
		name = queen_of_ashes_events.2.c #It appears the Devourer accepts no competition.
		trigger = {
			religion = amascut_religion
		}
		kill_ruler = yes
		add_government_power = {
			mechanic_type = enthrallment_mechanic
			power_type = enthrallment_power_1
			value = -20
		}
	}
}
