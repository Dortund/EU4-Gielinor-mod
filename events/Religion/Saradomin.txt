#The One, the Few, the Many
country_event = {
	id = saradomin_flavor.1
	title = saradomin_flavor.1.t
	desc = saradomin_flavor.1.d #The official position of the Church of Saradomin is that there are only three Gods: Saradomin, and the lesser gods Zamorak and Guthix. However, teachings vary. Some hold that there were other Gods that may have worked with Saradomin, and some argue that Saradomin alone is the sole God of Gielinor.
	picture = SARADOMIN_MISSIONARY_eventPicture
	
	is_triggered_only = yes

	trigger = {
		religion = saradomin
		NOT = { tag = ENT }
		NOT = { current_age = age_of_divination }
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = saradomin_flavor.1.a #Assert that Saradomin is the only true God.
		add_prestige = 10
		random_owned_province = {
			limit = {
				NOT = { religion = saradomin }
			}
			add_unrest = 10
		}
		
		ai_chance = {
			factor = 1
		}
	}
	option = {
		name = saradomin_flavor.1.b #Accept the godhood of the inferior Zamorak and Guthix.
		add_papal_influence = 3
		random_owned_province = {
			limit = {
				NOT = { religion = saradomin }
				NOT = { religion = guthix }
				NOT = { religion = zamorak }
			}
			add_unrest = 5
		}
		
		ai_chance = {
			factor = 1
		}
	}
	option = {
		name = saradomin_flavor.1.c #Imply that a myriad of Gods may exist, but they are unimportant.
		add_papal_influence = -3
		add_dip_power = 15
		random_owned_province = {
			limit = {
				NOT = { religion = saradomin }
			}
			add_unrest = -5
		}
		
		ai_chance = {
			factor = 1
		}
	}
}

#Monastic Privileges
country_event = {
	id = saradomin_flavor.3
	title = saradomin_flavor.3.t
	desc = saradomin_flavor.3.d #Monasteries can be the very model of order, with strict daily regimens and restricted living. In exchange, monasteries are typically granted freedom from taxes and from military recruitment, among other privileges.
	picture = SARADOMIN_MISSIONARY_eventPicture
	
	is_triggered_only = yes

	trigger = {
		religion = saradomin
		has_estate = estate_church
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = saradomin_flavor.3.a #Reassert the freedoms of the monasteries.
		add_papal_influence = 5
		add_estate_loyalty = {
			estate = estate_church
			loyalty = 10
		}
		ai_chance = {
			factor = 5
			modifier = {
				factor = 2
				NOT = {
					estate_loyalty = {
						estate = estate_church
						loyalty = 30
					}
				}
			}
		}
	}
	option = {
		name = saradomin_flavor.3.b #But what if we *really* need the money?

		add_papal_influence = -5
		add_estate_loyalty = {
			estate = estate_church
			loyalty = -10
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2
				num_of_loans = 1
			}
			modifier = {
				factor = 4
				num_of_loans = 3
			}
		}
	}
}

#Cardinals in Government
country_event = {
	id = saradomin_flavor.7
	title = saradomin_flavor.7.t
	desc = saradomin_flavor.7.d #With a cardinal present in [Root.GetName] we hold some influence over the Church of Saradomin, and they themselves hold some influence over us. We could tighten this relationship by granting the Cardinal an official governmental position.
	picture = SARADOMIN_MISSIONARY_eventPicture
	
	is_triggered_only = yes

	trigger = {
		religion = saradomin
		has_cardinal = yes
		NOT = { has_country_modifier = cardinal_in_administration }
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = saradomin_flavor.7.a #Yes, bring the Cardinal into government.
		if = {
			limit = {
				has_estate = estate_church
				NOT = {
					has_estate_influence_modifier = {
						estate = estate_church
						modifier = EST_VAL_CARDINAL_IN_ADMINISTRATION
					}
				}
			}
			add_estate_influence_modifier = {
				estate = estate_church
				desc = EST_VAL_CARDINAL_IN_ADMINISTRATION
				influence = 5
				duration = 7300
			}
		}
		add_country_modifier = {
			name = "cardinal_in_administration"
			duration = 7300
		}
		ai_chance = {
			factor = 1
		}
	}
	option = {
		name = saradomin_flavor.7.b #We can handle governance, let the Cardinal work on spiritual matters.
		
		add_prestige = 10
		ai_chance = {
			factor = 1
		}
	}
}

#Protection of Church Property
country_event = {
	id = saradomin_flavor.8
	title = saradomin_flavor.8.t
	desc = saradomin_flavor.8.d #Churches are sacred land, and many of the more remote churches of Saradomin are granted large surrounding estates with the intention of minimising distractions from worship. The limits of an church estate in [saradomin_protection_of_church_target.GetName] have been contested by local secular landholders who claim the borders of their territory are being infringed by the church.
	picture = SARADOMIN_MISSIONARY_eventPicture
	goto = saradomin_protection_of_church_target
	
	is_triggered_only = yes

	trigger = {
		religion = saradomin
		any_owned_province = {
			OR = {
				has_building = temple
				has_building = cathedral
			}
			religion = saradomin
		}
		has_estate = estate_church
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	immediate = {
		random_owned_province = {
			limit = {
				religion = saradomin
				OR = {
					has_building = temple
					has_building = cathedral
				}
			}
			save_event_target_as = saradomin_protection_of_church_target
		}
	}
	
	option = {
		name = saradomin_flavor.8.a #Side with the church.
		event_target:saradomin_protection_of_church_target = {
			add_province_modifier = {
				name = saradomin_expanded_church_lands
				duration = 7300
			}
		}
		change_estate_land_share = {
			estate = estate_church
			share = 2
		}
		add_estate_loyalty = {
			estate = estate_church
			loyalty = 10
		}
		ai_chance = {
			factor = 1
		}
	}
	option = {
		name = saradomin_flavor.8.b #Side with the secular landowners.
		event_target:saradomin_protection_of_church_target = {
			add_province_modifier = {
				name = saradomin_contracted_church_lands
				duration = 7300
			}
		}
		add_estate_loyalty = {
			estate = estate_church
			loyalty = -10
		}
		change_estate_land_share = {
			estate = estate_church
			share = -2
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0.5
				NOT = {
					estate_loyalty = {
						estate = estate_church
						loyalty = 30
					}
				}
			}
		}
	}
}
