country_event = {
	id = flavor_kds.7
	title = flavor_kds.7.t #Future of the North Kharidians
	desc = flavor_kds.7.d #It is time to decide our path.
	picture = ALKHARID_CITY_eventPicture
	
	is_triggered_only = yes
	fire_only_once = yes
	
	option = {
		name = flavor_kds.7.a #We are no longer defined by the Empire of Kharid!
		set_country_flag = KDS_al_kharid_missions_branch_flag
		swap_non_generic_missions = yes
		if = {
			limit = {
				is_subject_of_type = tributary_state
			}
			set_country_flag = KDS_breaking_free_flag
			overlord = {
				add_opinion = {
					who = ROOT
					modifier = kds_broke_from_empire_of_kharid
				}
				country_event = {
					id = flavor_kds.9
				}
			}
			grant_independence = yes
		}
		if = {
			limit = {
				NOT = { government = monarchy }
				is_revolutionary = no
			}
			change_government_to_monarchy = yes
		}
		add_government_reform = al_kharid_emirate
		country_event = {
			id = flavor_kds.10
		}
		ai_chance = {
			factor = 10
		}
	}
	option = {
		name = flavor_kds.7.b #We are the hand of the Pharoah!
		set_country_flag = KDS_loyalist_missions_branch_flag
		swap_non_generic_missions = yes
		random_country = {
			limit = {
				is_emperor_of_china = yes
			}
			add_historical_friend = ROOT
			ROOT = {
				add_historical_friend = PREV
			}			
		}
		remove_historical_friend = MST
		MST = {
			remove_historical_friend = ROOT
		}
		if = {
			limit = {
				alliance_with = MST
			}
			break_alliance = MST
		}
		add_government_reform = hand_of_het
		country_event = {
			id = flavor_kds.11
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0
				exists = MST
			}
		}
	}
}

					is_empty = yes

				}
