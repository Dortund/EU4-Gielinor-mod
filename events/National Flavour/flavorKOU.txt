country_event = {
	id = flavor_kou.3
	title = flavor_kou.3.t #The Royal Accord of Twill
	desc = flavor_kou.3.d #Relations between the people of Kourend and Varlamore have been uneasy since their monarch betrayed our people, but [Root.Monarch.GetTitle] [Root.Monarch.GetName] has an idea to help repair those relations - loan their Grand Museum the Royal Accord of Twill. This document was crucial for defining the power of Kourend's Houses and shall serve as a symbol of friendship.
	picture = KOUREND_OVERVIEW_eventPicture
	
	trigger = {
		tag = KOU
		NOT = { is_at_war = yes }
		VAR = {
			exists = yes
			has_country_flag = VAR_end_of_betrayal_flag
			NOT = { has_reform = tarnished_monarchy }
			NOT = {	truce_with = ROOT }
			owns_core_province = 525
		}
	}
	
	fire_only_once = yes

	mean_time_to_happen = {
		months = 400
		modifier = {
			factor = 0.2
			VAR = {
				has_country_flag = VAR_royal_accord_of_twill_gift_potential
			}
		}
	}
	
	option = {
		name = flavor_kou.3.a #Send the Royal Accord of Twill!
		random_list = {
			50 = {
				country_event = {
					id = flavor_kou.4
					days = 40
				}
			}
			50 = {
				VAR = {
					country_event = {
						id = flavor_kou.5
						days = 60
					}
				}
			}
		}
		ai_chance = {
			factor = 10
		}
	}
	option = {
		name = flavor_kou.3.b #We will not risk this crucial document to people we cannot trust.
		add_prestige = -5
		ai_chance = {
			factor = 0
		}
	}
}
