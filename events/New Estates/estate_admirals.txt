country_event = {
	id = admirals_estate_events.5
	title = admirals_estate_events.5.t
	desc = {
	   trigger = { NOT = { religion = xautak } }
	   desc = admirals_estate_events.5.da
	}
	desc = {
	   trigger = { religion = xautak }
	   desc = admirals_estate_events.5.db
	}
	picture = {
	   trigger = {
			NOT = { religion = xautak }
	   }
	   picture = ADMIRAL_ESTATE_eventPicture
	}
	picture = {
	   trigger = { religion = xautak }
	   picture = TERROR_OF_THE_LEVIATHAN_eventPicture
	}
	
	trigger = {
		has_estate = estate_admirals
		NOT = {
			estate_loyalty = {
				estate = estate_admirals
				loyalty = 30
			}
		}
		estate_territory = { 
			estate = estate_admirals
			territory = 10
		}
		any_owned_province = {
			OR = {
				development = 10
				province_trade_power = 5
			}
			is_capital = no
			is_city = yes
			has_port = yes
		}
	}
	
	is_triggered_only = yes
	
	immediate = {
		hidden_effect = {
			while = {
				limit = {
					NOT = { 
						development_in_provinces = {
							has_province_flag = port_protests
							value = estate
							estate = estate_admirals
						}
					}
					any_owned_province = {
						OR = {
							development = 10
							province_trade_power = 5
						}
						is_capital = no
						is_city = yes
						has_port = yes
						NOT = { has_province_flag = port_protests }
					}
					check_reducing_estate_revolt_size_trigger = {
						FLAG = port_protests
					}
				}
				random_owned_province = {
					limit = {
						OR = {
							development = 10
							province_trade_power = 5
						}
						is_capital = no
						is_city = yes
						has_port = yes
						NOT = { has_province_flag = port_protests }
					}
					set_province_flag = port_protests
				}
			}
		}
	}
	
	mean_time_to_happen = {
		days = 1
		modifier = {
			factor = 2.0
			NOT = {
				estate_loyalty = {
					estate = estate_admirals
					loyalty = 15
				}
			}
		}
	}
	
	option = {
		name = admirals_estate_events.5.a #
		ai_chance = {
			factor = 0
		}
		every_owned_province = {
			limit = {
				has_province_flag = port_protests
			}
			clr_province_flag = port_protests
			spawn_rebels = {
				type = particularist_rebels
				size = 1
			}
		}
	}
	option = {
		name = admirals_estate_events.5.b #
		ai_chance = {
			factor = 1
		}
		every_owned_province = {
			limit = {
				has_province_flag = port_protests
			}
			clr_province_flag = port_protests
			add_local_autonomy = 33
		}
		add_estate_loyalty = {
			estate = estate_admirals
			loyalty = 15
		}
	}
}
