#New Workshops
country_event = {
	id = workers_estate_events.8
	title = workers_estate_events.8.t
	desc = workers_estate_events.8.d #
	picture = WORKER_ESTATE_eventPicture
	goto = workshop_incoming
	
	is_triggered_only = yes
	
	trigger = {
		adm_tech = 6
		has_estate = estate_workers
		estate_loyalty = {
			estate = estate_workers
			loyalty = 75
		}		
		estate_influence = {
			estate = estate_workers
			influence = 75
		}
		OR = {
			AND = {
				adm_tech = 24
				any_owned_province = {
					NOT = {
						has_building = counting_house
					}
					OR = {
						has_building = workshop
						num_free_building_slots = 1
					}
				}
			}
			any_owned_province = {
				NOT = {
					has_building = workshop
				}
				num_free_building_slots = 1
			}
		}
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					adm_tech = 24
				}
				random_owned_province = {
					limit = {
						NOT = {
							has_building = counting_house
						}
						OR = {
							has_building = workshop
							num_free_building_slots = 1
						}
					}
					save_event_target_as = workshop_incoming
				}
			}
			else = {
				random_owned_province = {
					limit = {
						NOT = {
							has_building = workshop
						}
						num_free_building_slots = 1
					}
					save_event_target_as = workshop_incoming
				}
			}
		}
	}
	
	option = {
		name = workers_estate_events.8.a #
		event_target:workshop_incoming = {
			if = {
				limit = {
					has_building = workshop
				}
				remove_building = workshop
				add_building = counting_house
			}
			else = {
				add_building = workshop
			}
			add_base_production = 2
		}
	}
}
