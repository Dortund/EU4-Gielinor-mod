		}
		random_owned_province = {
			limit = {
				has_active_triggered_province_modifier = wildy_border_level_nine
				NOT = { culture = zombie }
			}
			change_culture = zombie
		}
		add_prestige = -25
	}

country_event = {
	id = wildernessborder.401
	title = wildernessborder.401.t # An Outlaw on the Wilderness Border
	desc = wildernessborder.401.d # The Wilderness is virtually impossible to police, and criminals who can brave the horrors within frequently cross the border in and out to evade justice. One such outlaw is known to prowl the border of [wildyborder401_target.GetName] and local people demand justice.
	picture = WILDERNESS_BANDIT_CAMP_eventPicture
	
	trigger = {
		any_owned_province = {
			has_active_triggered_province_modifier = wildy_border_level_one
		}
		any_owned_province = {
			current_institution = 100
		}
		any_country = {
			capital_scope = {
				region = forinthry_region
			}
			technology_group = northerntechgroup
			any_owned_province = {
				NOT = { current_institution = 99 }
			}
		}
	}
	
	immediate = {
		random_owned_province = {
			limit = {
				has_active_triggered_province_modifier = wildy_border_level_one
			}
			save_event_target_as = wildyborder401_target
		}
	}

	mean_time_to_happen = {
		months = 240 #Do not scale to national development; the longer a Wildy border you have, the more events you should get!
	}
	
	option = {
		name = wildernessborder.401.a #Keep the border safe and ensure the outlaw cannot return.
		add_mil_power = -5
		custom_tooltip = wildernessoutlaw_solo_spreadinstitution #The outlaw will find a home in the Wilderness, bringing knowledge of our institutions with them.
		random_country = {
			limit = {
				capital_scope = {
					region = forinthry_region
				}
				technology_group = northerntechgroup
				any_owned_province = {
					NOT = { current_institution = 99 }
				}
			}
			random_owned_province = {
				limit = {
					NOT = { current_institution = 99 }
				}
				add_next_institution_embracement = 10
			}
			country_event = { 
				id = wildernessborder.410
				days = 30
				random = 15
			}
		}
	}
	option = {
		name = wildernessborder.401.b #Sooner or later their reckless lifestyle will kill the outlaw.
		add_years_of_income = -0.1
		event_target:wildyborder401_target = {
			add_devastation = 5
		}
	}
}
