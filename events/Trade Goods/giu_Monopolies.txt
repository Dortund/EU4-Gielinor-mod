########################### 1+2: Admirals

# Extension of Admirals Fish Monopoly
country_event = {
	id = giu_monopoly_privileges.1
	title = giu_monopoly_privileges.1.t
	desc = giu_monopoly_privileges.1.d
	picture = ADMIRAL_ESTATE_eventPicture

	is_triggered_only = yes

	immediate = {
		remove_estate_monopoly_modifier = { privilege = estate_admirals_fish_monopoly }
	}
	
	option = {
		name = giu_monopoly_privileges.1.a
		reapply_estate_privilege = { privilege = estate_admirals_fish_monopoly }
	}
	option = {
		name = giu_monopoly_privileges.1.b
		custom_tooltip = giu_monopoly_privileges.1.b.tooltip

		every_owned_province = {
			limit = {
				trade_goods = fishing_spot
			}
			apply_estate_monopoly_modifier = { privilege = estate_admirals_fish_monopoly }
		}
	}
}
